# Color Prototype - 필요한 환경변수

# ===== 필수 환경변수 =====

# YouTube Data API (댓글 수집에 필요)
YOUTUBE_DATA_API_KEY=your_youtube_api_key_here
# 발급 방법: https://console.cloud.google.com/apis/credentials


# ===== 선택적 환경변수 =====

# 네이버 검색 API (팩트체크용, 선택사항)
NAVER_CLIENT_ID=your_naver_client_id
NAVER_CLIENT_SECRET=your_naver_client_secret
# 발급 방법: https://developers.naver.com/apps/#/register

# Ollama 설정 (로컬 LLM 사용 시)
OLLAMA_URL=http://localhost:11434
OLLAMA_MODEL=llama3.2
# Ollama 설치: https://ollama.ai/


# ===== 환경변수 설정 방법 =====

# 1. 프로젝트 루트에 .env 파일 생성
# 2. 위 환경변수들을 복사하여 .env 파일에 붙여넣기
# 3. your_xxx_here 부분을 실제 값으로 변경

# 예시 (.env 파일):
# YOUTUBE_DATA_API_KEY=AIzaSyABC123DEF456GHI789JKL012MNO345PQR
# NAVER_CLIENT_ID=abc123def456
# NAVER_CLIENT_SECRET=xyz789uvw012
# OLLAMA_URL=http://localhost:11434
# OLLAMA_MODEL=llama3.2


# ===== 주의사항 =====

# - .env 파일은 절대 Git에 커밋하지 마세요!
# - .gitignore에 .env 추가 권장
# - API 키는 외부에 노출되지 않도록 주의하세요
